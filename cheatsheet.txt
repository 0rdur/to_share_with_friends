/* 2. OPERATOR CATEGORIES & SYNTAX EXAMPLES */

// Arithmetic

a - b          // Subtraction
a * b          // Multiplication
a / b          // Division (truncates if both operands are integers)
a % b          // Modulo / Remainder (integers only)

// Increment / Decrement
a++            // Post-increment: returns 'a', then increments 'a'
++a            // Pre-increment : increments 'a', then returns 'a'

a == b         // Equal

!a             // Logical NOT (1 if a==0, else 0)

#include <stdio.h>

/* --- 1. PROTOTYPES (DECLARATIONS) --- */

// Value & Pointer parameters
void passByValue(int num);
void passByPointer(int *numPtr);

// Array parameters (Both syntaxes are identical to the compiler)
void printArray(const int arr[], int size);   // Read-only (const)
void modifyArray(int *arr, int size);         // Read-Write

// String parameters
void printString(const char str[]);           // Read-only string
void modifyString(char *str);                 // Read-write string


/* --- 2. USAGE IN MAIN --- */

int main(void) {
    int x = 10;
    int numbers[3] = {1, 2, 3};
    char greeting[] = "Hello";

    // Variables
    passByValue(x);       // 'x' remains 10
    passByPointer(&x);    // 'x' becomes 20 (passed memory address)

    // Arrays (Name of array acts as address of first element)
    printArray(numbers, 3);
    modifyArray(numbers, 3); // Modifies original array

    // Strings
    printString(greeting);
    modifyString(greeting);  // Modifies original string

    return 0;
}


/* --- 3. DEFINITIONS --- */

// 1. Pass-by-Value: Operates on a copy
void passByValue(int num) {
    num = 99; // Does NOT change 'x' in main
}

// 2. Pass-by-Pointer: Operates on original variable
void passByPointer(int *numPtr) {
    *numPtr = 20; // Dereferences pointer to modify 'x' in main
}

// 3. Read-Only Array: Must pass size separately!
void printArray(const int arr[], int size) {
    for (int i = 0; i < size; i++) {
        printf("%d ", arr[i]);
    }
    printf("\n");
}

// 4. Modify Array: Changes affect the caller's array
void modifyArray(int *arr, int size) {
    for (int i = 0; i < size; i++) {
        arr[i] *= 2; // Doubles each element in main's array
    }
}

// 5. Read-Only String: No size parameter needed due to '\0'
void printString(const char str[]) {
    printf("%s\n", str);
}

// 6. Modify String
void modifyString(char *str) {
    str[0] = 'h'; // Changes "Hello" to "hello" in main
}

/* ==========================================================================
   C LOOPS CHEATSHEET
   ========================================================================== */

// 1. FOR LOOP (Count-controlled loop)
//    for (init; condition; update)
for (int i = 0; i < 5; i++) {
    printf("%d ", i); // Prints: 0 1 2 3 4
}


// 2. WHILE LOOP (Pre-condition check: 0 or more runs)
int i = 0;
while (i < 5) {
    printf("%d ", i); // Prints: 0 1 2 3 4
    i++;
}


// 3. DO-WHILE LOOP (Post-condition check: GUARANTEED at least 1 run)
int j = 0;
do {
    printf("%d ", j); // Prints: 0 (even if condition starts as false)
    j++;
} while (j < 0);      // Note the closing semicolon!


// 4. CONTROL FLOW (break & continue)
for (int k = 0; k < 10; k++) {
    if (k == 2) continue; // Skips iteration when k == 2 (goes straight to k++)
    if (k == 5) break;    // Immediately terminates loop when k == 5
    printf("%d ", k);     // Prints: 0 1 3 4
}


// 5. INFINITE LOOPS
for (;;) { /* runs forever */ }
while (1) { /* runs forever */ }
